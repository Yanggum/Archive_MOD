scripted_gui = {
	story_archive_gui = {
		
		context_type = player_context
		
		window_name = "story_archive_gui_window"

		visible = {
			ROOT = {
				OR = {
					check_variable = {
						var = story_archive
						value = 1
					}
				}
				
			}

			#has_variable = BEL_story_archive			
		}

		dynamic_lists = {
			story_archive_grid_box = {
				array = story_archive_national_array

				#value of the current array index. Optional. Default value is v.
				value = story_arc_value

				#current array index. Optional. Default value is i.
				index = story_arc_index
				
				#This will change the scope to the value if yes
				#change_scope = no

				#The GUI drawn for each index
				entry_container = "story_archive_gui_array"
			}
		}

		effects = {
			#Example Buttons to show it working
			btn_close_click = {
				set_variable = {
					story_archive = 0
				}
			}

			"btn_[?story_arc_value]_archive_[?story_arc_index]_click" = {
				set_variable = {
					story_event_var = story_arc_index
				}
				log = "story_arc_value = [?story_arc_value]"
				log = "story_arc_index = [?story_arc_index]"
				add_to_variable = {
					var = story_event_var 
					value = 1
				}
			}
		}
	}
	story_archive_open_button_gui = {
		context_type = player_context
		parent_window_token = top_bar
		
		window_name = "story_archive_open_button_gui_window"

		visible = {	
			ROOT = {
				OR = {
					tag = GEK
					tag = TRM
					tag = BEL
					tag = KSP
					tag = HEU
				}
			}	
		}

		effects = {
			#Example Buttons to show it working
			story_archive_open_click = {
				BA_storyarchive_openui_effects = yes
				if = {
					limit = {
						NOT = {
							check_variable = {
								story_archive = 1
							}
						}
					}
					set_variable = {
						story_archive = 1
					}
				}
				else_if = {
					limit = {
						NOT = {
							check_variable = {
								story_archive = 0
							}
						}
					}
					set_variable = {
						story_archive = 0
					}
				}
			}
		}
	}
}
