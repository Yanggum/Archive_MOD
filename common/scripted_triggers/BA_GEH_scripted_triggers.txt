	if = {
		limit = {
			check_variable = {
				GEM_ibuki_system_lv_value_temp = 1
			}
		}
		set_temp_variable = {
			GEM_ibuki_system_lv_value_return = 19
		}
	}
	else_if = {
		limit = {
			check_variable = {
				GEM_ibuki_system_lv_value_temp = 2
			}
		}
		set_temp_variable = {
			GEM_ibuki_system_lv_value_return = 39
		}
	}
	else_if = {
		limit = {
			check_variable = {
				GEM_ibuki_system_lv_value_temp = 3
			}
		}
		set_temp_variable = {
			GEM_ibuki_system_lv_value_return = 59
		}
	}
	else_if = {
		limit = {
			check_variable = {
				GEM_ibuki_system_lv_value_temp = 4
			}
		}
		set_temp_variable = {
			GEM_ibuki_system_lv_value_return = 79
		}
	}
	else_if = {
		limit = {
			check_variable = {
				GEM_ibuki_system_lv_value_temp = 5
			}
		}
		set_temp_variable = {
			GEM_ibuki_system_lv_value_return = 99
		}
	}
	else = {
		set_temp_variable = {
			GEM_ibuki_system_lv_value_return = 100
		}
	}
}
