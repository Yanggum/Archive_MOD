		limit = { has_country_flag = IRE_government_FF has_country_flag = IRE_FF_with_LP }
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2
			check_variable = { IRE_FF_plus_LP > 100 }
		}
	}
	else_if = {
		limit = { has_country_flag = IRE_government_FF has_country_flag =  IRE_FF_with_FG }
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2
			check_variable = { IRE_FF_plus_FG > 100 }
		}
	}
	else_if = {
		limit = {
			has_country_flag = IRE_government_FF
		}
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2
			check_variable = { IRE_parliament_seat_array^1 > 100 }
		}
	}
	else_if = {
		limit = {
			has_country_flag = IRE_government_LP
		}
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2
			check_variable = { IRE_parliament_seat_array^0 > 100 }
		}
	}
	else_if = {
		limit = {
			has_country_flag = IRE_government_FG
		}
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2

		}
	}
	else = {

			tooltip = IRE_LT_tt_2
			always = no
		}
	}
