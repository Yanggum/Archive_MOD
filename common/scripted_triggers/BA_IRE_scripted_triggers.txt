IRE_majority = {
	if = {
		limit = { has_government = democratic has_country_flag = IRE_FF_with_LP }
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2
			check_variable = { IRE_FF_plus_LP > 100 }
		}
	}
	else_if = {
		limit = { has_government = democratic has_country_flag =  IRE_FF_with_FG }
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2
			check_variable = { IRE_FF_plus_FG > 100 }
		}
	}
	else_if = {
		limit = {
			has_government = democratic
		}
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2
			check_variable = { IRE_parliament_seat_array^1 > 100 }
		}
	}
	else_if = {
		limit = {
			has_government = communism
		}
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2
			check_variable = { IRE_parliament_seat_array^0 > 100 }
		}
	}
	else_if = {
		limit = {
			has_government = neutrality
		}
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2
			check_variable = { IRE_parliament_seat_array^2 > 100 }
		}
	}
	else = {
		custom_trigger_tooltip = {
			tooltip = IRE_LT_tt_2
			always = no
		}
	}
}
