defined_text = {
	name = GetTrCFLevelText
	text = {
		trigger = {
			OR = {
				has_dynamic_modifier = { modifier = TRM_appeasement_germany_modifier }
				has_dynamic_modifier = { modifier = TRM_pro_german_propaganda_modifier }
				has_dynamic_modifier = { modifier = TRM_trinity_social_issues_modifier }
			}
			check_variable = { var = TRM_pro_german_propaganda_conscription_factor value = -0.50 compare = less_than_or_equals }
		}
		localization_key = TRM_max_CF_tt
	}
	text = {
		trigger = {
			OR = {
				has_dynamic_modifier = { modifier = TRM_appeasement_germany_modifier }
				has_dynamic_modifier = { modifier = TRM_pro_german_propaganda_modifier }
				has_dynamic_modifier = { modifier = TRM_trinity_social_issues_modifier }
			}
			check_variable = { TRM_pro_german_propaganda_conscription_factor > -0.50 }
		}
		localization_key = TRM_increasing_cf_tt 
	}
}
