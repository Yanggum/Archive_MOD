TRK_eden_collapse_decision_category = {
	
	TRK_eden_collapse_decision = {
		icon = GFX_decision_ger_reichskommissariats

		fire_only_once = yes

		ai_will_do = {
			base = 4000 # 반드시 찍는다
		}

		allowed = {
			NOT = { tag = TRK }
			NOT = { tag = GEH }
			NOT = { tag = HEU }
			NOT = { tag = GEM }
		}

		visible = {
			AND = {
				OR = {	
					# 룩셈부르크가 병합되었거나 괴뢰국
					NOT = { country_exists = HEU }
					HEU = { is_puppet_of = ROOT }
				}
				OR = {	
					# 게헨나 선도부가 병합되었거나 괴뢰국
					NOT = { country_exists = GEH }
					GEH = { is_puppet_of = ROOT }
				}
				OR = {	
					# 게헨나 판데모니움 소사이어티가 병합되었거나 괴뢰국
					NOT = { country_exists = GEM }
					GEM = { is_puppet_of = ROOT }
				}
			}
			# 위 조건이 충족된 상태(AND)로 에덴 관리기구가 존재
			country_exists = TRK
			# 게헨나 중앙구 또는 남구를 점령한 국가에게 보임
			OR = {
				owns_state = 55
				owns_state = 52
			}
		}

		available = {
			is_subject = no
			TRK = { is_subject = no }
			NOT = { has_war_with = TRK }
			# 게헨나 중앙구와 남구 점령국가가 병합 가능
			owns_state = 55
			owns_state = 52
		}

		complete_effect = {
			TRK = {
				every_owned_state = {
					limit = { is_core_of = TRK }
					remove_core_of = TRK
				}
			}
			annex_country = {
				target = TRK # 에덴 관리기구 죽임
			}
		}
	}

}
ARI_gehenna_influence_struggle_category = {

	ARI_gehenna_influence_struggle = {
		icon = GFX_decision_SWI_consolidate_council_power
		name = TRK_gehenna_influence_struggle
		days_mission_timeout = 280
		priority = 1

        allowed = {
            tag = ARI
        }

		activation = {
			always = no
		}

		available = {
			always = no
    	}

		cancel_trigger = {
			OR = {
				NOT = { country_exists = ARI }
				NOT = { country_exists = TRK }
			}
			hidden_trigger = {
				has_global_flag = ARI_gehenna_influence_struggle_stop_flag
			}
		}

		cancel_effect = {
			set_global_flag = ARI_gehenna_influence_struggle_stop_flag
		}

		timeout_effect = {
			hidden_effect = {
				if = {
					limit = {
						check_variable = {
							global.ARI_gehenna_influence_struggle < global.TRK_gehenna_influence_struggle
						}
					}
					var:global.TRK_owner?GEM = {
						diplomatic_relation = {
							country = TRK
							relation = military_access
						}
						declare_war_on = {
							target = ARI
							type = annex_everything
						}

						TRK = {
							diplomatic_relation = {
								country = PREV
								relation = military_access
							}
							declare_war_on = {
								target = ARI
								type = annex_everything
							}
						}
					}
				}
				else = {
					ARI = {
						create_wargoal = {
							target = TRK
							type = annex_everything
						}
					}
				}
			}
			effect_tooltip = {
				custom_effect_tooltip = TRK_gehenna_influence_struggle_ARI
				ARI = {
					create_wargoal = {
						target = TRK
						type = annex_everything
					}
				}
				custom_effect_tooltip = enter_tt
				custom_effect_tooltip = TRK_gehenna_influence_struggle_TRK
				var:global.TRK_owner?GEM = {
					diplomatic_relation = {
						country = TRK
						relation = military_access
					}
					declare_war_on = {
						target = ARI
						type = annex_everything
					}

					TRK = {
						diplomatic_relation = {
							country = PREV
							relation = military_access
						}
						declare_war_on = {
							target = ARI
							type = annex_everything
						}
					}
				}
			}
		}
	}
	
	ARI_gehenna_influence_struggle_action = {
		icon = GFX_decision_generic_prepare_civil_war
		name = TRK_gehenna_influence_struggle_action

		ai_will_do = {
			factor = 10
			modifier = {
				add = 10
				has_war = no
			}
		}

		available = {
			has_active_mission = ARI_gehenna_influence_struggle
		}

		visible = {
			has_active_mission = ARI_gehenna_influence_struggle
		}

		cost = 35
		
		days_remove = 15

		modifier = {
			political_power_cost = 0.15
		}

		complete_effect = {
			add_to_variable = {
				global.ARI_gehenna_influence_struggle = 5
				tooltip = ARI_gehenna_influence_struggle_action_tt
			}
		}
		
		remove_effect = {
		}
	}
}
