				}
				else_if = {
					limit = {
						check_variable = { IRE_FF_sup_ > 0 }

					add_to_variable = { IRE_FF_sup_ = -1 }
				}
				else_if = {
					limit = {
						check_variable = { IRE_LP_sup_ > 0 }
					}
					add_to_variable = { IRE_LP_sup_ = -1 }
				}
			}
			else_if = {
				limit = {
					check_variable = { plus_support > 200 }
				}
				if = {
					limit = {
						check_variable = { IRE_FG_sup_ > 0 }
					}
					add_to_variable = { IRE_FG_sup_ = -1 }
				}
				else_if = {
					limit = {
						check_variable = { IRE_FF_sup_ > 0 }
					}
					add_to_variable = { IRE_FF_sup_ = -1 }
				}
				else_if = {
					limit = {
						check_variable = { IRE_LP_sup_ > 0 }
					}
					add_to_variable = { IRE_LP_sup_ = -1 }
				}
			}
			if = {
				limit = {
					check_variable = { plus_support < 200 }
				}
				add_to_variable = { IRE_N_support = 1 }
			}
			else_if = {
				limit = {
					check_variable = { plus_support < 200 }
				}
				add_to_variable = { IRE_N_support = 1 }
			}
			set_variable = {
				var = IRE_parliament_seat_array^0
				value = IRE_LP_sup_
			}
			set_variable = {
				var = IRE_parliament_seat_array^1
				value = IRE_FF_sup_
			}
			set_variable = {
