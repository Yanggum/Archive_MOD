KPD_add_or_modify_politburo = {
	if = {
		limit = {
			NOT = { has_dynamic_modifier = { modifier = KPD_politburo_modifier }}
		}
		custom_effect_tooltip = GER_add_politburo_modifier_tt
		hidden_effect = {
			add_dynamic_modifier = { modifier = KPD_politburo_modifier } 
		}
	}
	else = {
		custom_effect_tooltip = KPD_modify_politburo_modifier_tt
	}
}

KPD_modify_army = {
	if = {
		limit = {
			NOT = {
				has_dynamic_modifier = { modifier = KPD_army_dynamic_modifier }
				has_dynamic_modifier = { modifier = KPD_the_red_army_dynamic_modifier }
				has_dynamic_modifier = { modifier = KPD_prussian_army_dynamic_modifier }
				has_dynamic_modifier = { modifier = KPD_national_defense_force_dynamic_modifier }
				has_dynamic_modifier = { modifier = KPD_citizens_in_uniform_dynamic_modifier }
				has_dynamic_modifier = { modifier = KPD_national_peoples_army_dynamic_modifier }
			}
		}
		custom_effect_tooltip = KPD_add_army_dynamic_modifier_tt
		hidden_effect = {
			add_dynamic_modifier = { modifier = KPD_army_dynamic_modifier } 
		}
	}
	else_if = {
		limit = {
			has_dynamic_modifier = { modifier = KPD_army_dynamic_modifier }
		}
		custom_effect_tooltip = KPD_army_dynamic_modifier_tt
	}
	else_if = {
		limit = {
			has_dynamic_modifier = { modifier = KPD_the_red_army_dynamic_modifier }
		}
		custom_effect_tooltip = KPD_the_red_army_dynamic_modifier_tt
	}
	else_if = {
		limit = {
			has_dynamic_modifier = { modifier = KPD_prussian_army_dynamic_modifier }
		}
		custom_effect_tooltip = KPD_prussian_army_dynamic_modifier_tt
	}
	else_if = {
		limit = {
			has_dynamic_modifier = { modifier = KPD_national_defense_force_dynamic_modifier }
		}
		custom_effect_tooltip = KPD_national_defense_force_dynamic_modifier_tt
	}
	else_if = {
		limit = {
			has_dynamic_modifier = { modifier = KPD_citizens_in_uniform_dynamic_modifier }
		}
		custom_effect_tooltip = KPD_citizens_in_uniform_dynamic_modifier_tt
	}
	else_if = {
		limit = {
			has_dynamic_modifier = { modifier = KPD_national_peoples_army_dynamic_modifier }
		}
		custom_effect_tooltip = KPD_national_peoples_army_dynamic_modifier_tt
	}
}
