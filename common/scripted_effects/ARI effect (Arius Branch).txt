				}
			}
		}
	}
	hidden_effect = {
		set_country_flag = ARI_penal_units_raised_flag
	}
}

AASRU_scripted_effect_on_value_change = {
	set_temp_variable = {
		AASRU_origin_temp = AASRU_var
	}

	add_to_temp_variable = {
		AASRU_origin_temp = AASRU_temp
	}

	clamp_temp_variable = {
		var = AASRU_origin_temp
		min = 0
		max = 100
	}

	if = {
		limit = {
			has_country_leader_with_trait = ARI_saori_skepticism_about_reform
			check_variable = {
				AASRU_origin_temp > 49
			}
		}
		swap_ruler_traits = {
			remove = ARI_saori_skepticism_about_reform
			add = ARI_saori_trust_in_reform
		}
	}
	else_if = {
		limit = {
			has_country_leader_with_trait = ARI_saori_trust_in_reform
			check_variable = {
				AASRU_origin_temp < 50
			}
		}
		swap_ruler_traits = {
			remove = ARI_saori_trust_in_reform
			add = ARI_saori_skepticism_about_reform
		}
	}

	set_variable = {
		AASRU_var = AASRU_origin_temp
	}
}

	}

	clamp_variable = {
		var = MISAKI_var
		max = 100
		min = 0
	}

	### GUI 연산부
	set_temp_variable = {
		MISAKI_var_temp = MISAKI_var
	}
	set_temp_variable = {
		MISAKI_light_y_var_temp = 253
	}
	set_temp_variable = {
		MISAKI_light_y_var_mody_temp = MISAKI_var_temp
	}
	multiply_temp_variable = {
		MISAKI_light_y_var_mody_temp = -2.6	
	}
	add_to_temp_variable = {
		MISAKI_light_y_var_temp = MISAKI_light_y_var_mody_temp
	}
	set_variable = {
		MISAKI_light_y_var = MISAKI_light_y_var_temp
	}

	### 값에 대한 이벤트 격발부

	if = {
		limit = {
			has_completed_focus = ARI_everything_is_vain
			check_variable = {
				MISAKI_var < 51
			}
			NOT = { has_country_flag = MISAKI_var_50_flag }
			has_country_leader = {
				character = ARI_imashino_misaki
				ruling_only = yes
			}
		}
		set_country_flag = MISAKI_var_50_flag
		set_country_leader_description = {
			desc = GEH_misaki_imashino_mid_leader_desc
		}
	}

	if = {
		limit = {
			has_completed_focus = ARI_everything_is_vain
			check_variable = {
				MISAKI_var < 26
			}
			NOT = { has_country_flag = ARI_moderate_event_chain_2_flag }
			has_country_leader = {
				character = ARI_imashino_misaki
				ruling_only = yes
			}
		}
		set_country_flag = ARI_moderate_event_chain_2_flag
		country_event = ARI_moderate_event_chain.2
		set_country_leader_description = {
			desc = GEH_misaki_imashino_late_leader_desc
		}
	}

	if = {
		limit = {
			has_completed_focus = ARI_everything_is_vain
			check_variable = {

			}

	}
