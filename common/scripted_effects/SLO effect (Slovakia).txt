SLO_economy_chaos_down = {
	if = {
		limit = {
			has_idea = SLO_economy_chaos_idea1
		}
		swap_ideas = {
			remove_idea = SLO_economy_chaos_idea1
			add_idea = SLO_economy_chaos_idea2
		}
	}
	else_if = {
		limit = {
			has_idea = SLO_economy_chaos_idea2
		}
		swap_ideas = {
			remove_idea = SLO_economy_chaos_idea2
			add_idea = SLO_economy_chaos_idea3
		}
	}
	else_if = {
		limit = {
			has_idea = SLO_economy_chaos_idea3
		}
		remove_ideas = SLO_economy_chaos_idea3
	}
	else = {
		add_tech_bonus = {
			bonus = 1
			ahead_reduction = 2
			uses = 1
			category = industry
			category = electronics
		}
	}
}
