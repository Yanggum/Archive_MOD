		set_variable = { # 값 지정
			GEM_ibuki_system_two_array_value^2 = GEM_ibuki_system_temp_value
		}
	}
	else_if = {
		limit = {
			check_variable = {
				GEM_ibuki_system_array_temp_target = 4
			}
		}

			GEM_ibuki_system_temp_value = GEM_ibuki_system_two_array_value^3
		}
		add_to_temp_variable = {
			GEM_ibuki_system_temp_value = GEM_ibuki_system_array_temp_value
			tooltip = GEM_ibuki_system_array_value_4_tt
		}

		set_temp_variable = {
			GEM_ibuki_system_lv_value_temp = GEM_ibuki_system_two_array_lv^3
		}
		GEM_ibuki_system_lv_value_get = yes
		if = {
			limit = {
				check_variable = {
					GEM_ibuki_system_temp_value > GEM_ibuki_system_lv_value_return
				}
			}
			clamp_temp_variable = {
				var = GEM_ibuki_system_temp_value
				max = GEM_ibuki_system_lv_value_return
			}
			custom_effect_tooltip = GEM_ibuki_system_array_value_modify_clamp_tt
		}
		########### LV 확인
		set_variable = { # 값 지정
			GEM_ibuki_system_two_array_value^3 = GEM_ibuki_system_temp_value
		}
	}
