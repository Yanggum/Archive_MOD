CAN_change_par_anger = {
	if = {
		limit = {
			has_idea = CAN_parliament_anger_ns
			OR = {
				check_variable = { CAN_anger_change > 0 }
				AND = {
					check_variable = { CAN_anger_change < 0 }
					check_variable = { CAN_parliament_anger > 0 }
				}
			}
		}

		set_temp_variable = { CAN_anger_lower_limit = 1 }
		subtract_from_temp_variable = { CAN_anger_lower_limit = CAN_parliament_anger }

		set_temp_variable = { CAN_anger_upper_limit = 10 }
		subtract_from_temp_variable = { CAN_anger_upper_limit = CAN_parliament_anger }

		# avoid going < 1 or > 10
		clamp_temp_variable = {
			var = CAN_anger_change
			min = CAN_anger_lower_limit
			max = CAN_anger_upper_limit
		}

		if = {
			limit = {
				OR = {
					NOT = { has_variable = CAN_parliament_anger } #bypass the check when initialising
					NOT = { check_variable = { CAN_anger_change = 0 } }
				}
			}
			if = {
				limit = { check_variable = { CAN_anger_change > 0 } }
				custom_effect_tooltip = CAN_increase_par_anger
				
			}
			else = {
				custom_effect_tooltip = CAN_decrease_par_anger
				
			}

			# for the tooltips
			set_temp_variable = { CAN_par_anger_pp_temp = 0.1 }
			multiply_temp_variable = { CAN_par_anger_pp_temp = CAN_anger_change }

			set_temp_variable = { CAN_par_anger_cg_temp = 0.02 }
			multiply_temp_variable = { CAN_par_anger_cg_temp = CAN_anger_change }

			# set the new variables
			add_to_variable = { CAN_parliament_anger = CAN_anger_change }
			add_to_variable = { CAN_par_anger_pp = CAN_par_anger_pp_temp }
			add_to_variable = { CAN_par_anger_cg = CAN_par_anger_cg_temp }
		}
	}
}
