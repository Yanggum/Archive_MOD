on_actions = {

	on_startup = {
		effect = {
			every_country = {
				if = {
					limit = { tag = GEH }

					#선도부 풍기
					geh_discipline_system_setting_effect = yes
					add_dynamic_modifier = {
						modifier = geh_discipline_idea
					}

					# AI 선도부 루트 설정
					if = {
						limit = {
							has_game_rule = {
                        	    rule = GEH_ai_behavior
                        	    option = DEFAULT
                        	}
							tag = GEH
						}
						random_list = {
							10 = {
								set_variable = {
									GEH_route_random = 0 # 히나 독재정
								}
							}
							10 = {
								set_variable = {
									GEH_route_random = 1 # 이오리 루트
								}
							}
							10 = {
								set_variable = {
									GEH_route_random = 2 # 아코 루트
								}
							}
							10 = {
								set_variable = {
									GEH_route_random = 3 # 기본 루트
								}
							}
						}
					}
					else_if = {
						limit = {
							has_game_rule = {
                        	    rule = GEH_ai_behavior
                        	    option = GEH_HINA_AI_BEHAVIOR
                        	}
							tag = GEH
						}
						set_variable = {
							GEH_route_random = 0 # 히나 독재정
						}
					}
					else_if = {
						limit = {
							has_game_rule = {
                        	    rule = GEH_ai_behavior
                        	    option = GEH_IORI_AI_BEHAVIOR
                        	}
							tag = GEH
						}
						set_variable = {
							GEH_route_random = 1 # 이오리 루트
						}
					}
					else_if = {
						limit = {
							has_game_rule = {
                        	    rule = GEH_ai_behavior
                        	    option = GEH_AKO_AI_BEHAVIOR
                        	}
							tag = GEH
						}
						set_variable = {
							GEH_route_random = 2 # 아코 루트
						}
					}
					else_if = {
						limit = {
							has_game_rule = {
                        	    rule = GEH_ai_behavior
                        	    option = GEH_GENERIC_AI_BEHAVIOR
                        	}
							tag = GEH
						}
						set_variable = {
							GEH_route_random = 3 # 기본 루트
						}
					}

					### 선도부-판데모니움 소사이어티 동아리 이벤트 ###
					country_event = { id = GEH.GEM.shared_event.1 days = 380 } # 새해의 아페테리프
					country_event = { id = GEH.GEM.shared_event.2 days = 560 } # Trip-Trap-Train (+180일)
					country_event = { id = GEH.GEM.shared_event.3 days = 740 } # 군옥수수 팝니다 (+180일)
					country_event = { id = GEH.GEM.shared_event.4 days = 920 } # 의사당 밑에 온천이 있다고? (+180일)
				}
				if = {
					limit = { tag = GEM }

					### 선도부-판데모니움 소사이어티 동아리 이벤트 ###
					country_event = { id = GEH.GEM.shared_event.1 days = 380 } # 새해의 아페테리프
					country_event = { id = GEH.GEM.shared_event.2 days = 560 } # Trip-Trap-Train (+180일)
					country_event = { id = GEH.GEM.shared_event.3 days = 740 } # 군옥수수 팝니다 (+180일)
					country_event = { id = GEH.GEM.shared_event.4 days = 920 } # 의사당 밑에 온천이 있다고? (+180일)
				}
			}
		}
	}

	on_puppet = {
		effect = {
			if = {
				limit = {
					has_global_flag = GEH_managed_democracy_union_flag
					GEH = { is_subject = no }
					FROM = { tag = GEH }
				}
				add_ideas = GEH_managed_democracy_union_idea
			}
		}
	}

	on_release_as_puppet = {
		effect = {
			if = {
				limit = {
					has_global_flag = GEH_managed_democracy_union_flag
					GEH = { is_subject = no }
					FROM = { tag = GEH }
				}
				add_ideas = GEH_managed_democracy_union_idea
			}
		}
	}

}
