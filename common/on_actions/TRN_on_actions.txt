						rule = Trinity_TeaParty_swimsuit_Portrait_Rule

					TRM = { has_focus_tree = Trinity_focus }
				}

						add_trait = {
							slot = political_advisor
							trait = TRN_magician_of_the_old_library
						}
						remove_trait = {
							slot = political_advisor
							trait = TRN_economist_ui
						}
					}
					add_dynamic_modifier = { modifier = TRM_appeasement_germany_modifier }
					country_event = { id = trinity_flavor_events.36 hours = 01 }
					country_event = { id = trinity_events.6 days = 02 }
					country_event = { id = trinity_flavor_events.29 days = 04 }
				}
			}
		}
		effect = {
			if = {
				limit = {
					any_country = { has_idea = TRN_poland_trinity_border_dispute_ns }
				}
				TRM = {
					add_opinion_modifier = {
						target = POL
						modifier = TRN_poland_trinity_border_dispute_modifier
					}
				}

					add_opinion_modifier = {

					}
				}
			}

	}
	on_war = {
		effect = {
			if = { # 트리니티가 "TRM_war_preparations" 중점을 완료 했고 전쟁을 선포했을때
				limit = {
					original_tag = TRM
					TRM = { 
						OR = {
							has_completed_focus = TRM_war_preparations
							AND = {
								has_completed_focus = TRM_Expand_the_Logistics_Corps
								has_focus_tree = Trinity_focus
							}
						}
						NOT = { has_country_flag = TRM_had_strengthened_supply_lines_ns }
					}
				}
				TRM = {
					add_timed_idea = {
						idea = GER_strengthened_supply_lines_ns
						days = 730
					}
					set_country_flag = TRM_had_strengthened_supply_lines_ns
				}
			}
			if = { # 게헨나가 "GEM_war_preparations" 중점을 완료 했고 전쟁을 선포했을때
				limit = {
					original_tag = GEM
					GEM = { 
						OR = {
							has_completed_focus = GEM_war_preparations
						}
						NOT = { has_country_flag = GEM_had_strengthened_supply_lines_ns }
					}
				}
				GEM = {
					add_timed_idea = {
						idea = GER_strengthened_supply_lines_ns
						days = 730
					}
					set_country_flag = GEM_had_strengthened_supply_lines_ns
				}
			}
		}
		effect = {
			if = {
				limit = {
					original_tag = TRM
					TRM = { 
						has_completed_focus = TRN_supply_line_improvements 
						NOT = { has_country_flag = TRM_had_strengthened_supply_lines_ns }
					}
				}
				TRM = {
					add_timed_idea = {
						idea = GER_strengthened_supply_lines_ns
						days = 730
					}
					set_country_flag = TRM_had_strengthened_supply_lines_ns
				}
			}
		}
	}
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					ROOT = {
						original_tag = TRM
						has_idea = TRM_anti_gehenna_idea
					}
					FROM.FROM = {
						NOT = { is_core_of = TRM }
						BA_is_GER_core = yes
						OR = {
							is_core_of = GEM
							is_core_of = GEH
							is_core_of = HEU
						}
					}
				}
				log = "[GetDateText], [ROOT.GetName] control [FROM.FROM.GetName]"
				if = {
					limit = { 
						ROOT = { 
							check_variable = { 
								TRM_anti_geh_emotion_var = 0 
							}
						}
					}
					ROOT = {
						set_variable = { TRM_anti_geh_emotion_var = FROM.FROM }
						country_event = {
							id = TRMDIP.20
							hours = 0
						}
					}
				}
				
				FROM.FROM = {
					add_dynamic_modifier = { modifier = TRM_anti_geh_emotion }
				}
			}
		}
	}
	on_unit_leader_level_up = {
		effect = {
			if = {
				limit = {
					FROM = { tag = TRM }
					ROOT = { 
						skill > 2
						has_trait = trait_TRN_lack_of_experience 
					}
				}
				unit_leader_event = trinity_flavor_events.38
			}
		}
	}

		effect = {
			IF = {
				limit = {
					has_completed_focus = TRN_unprovable_problem
				}
				IF = {
					limit = {
						has_country_leader = {
							character = TRN_Kirifuji_Nagisa

						}
						TRN_Kirifuji_Nagisa = { has_trait = TRN_nagisa_1_off }
