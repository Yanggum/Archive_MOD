	on_war = {
		effect = {
			if = { # 트리니티가 "TRM_war_preparations" 중점을 완료 했고 전쟁을 선포했을때
				limit = {
					original_tag = TRM
					TRM = { 
						OR = {
							has_completed_focus = TRM_war_preparations
							AND = {
								has_completed_focus = TRM_Expand_the_Logistics_Corps
								has_focus_tree = Trinity_focus
							}
						}
						NOT = { has_country_flag = TRM_had_strengthened_supply_lines_ns }
					}
				}
				TRM = {
					add_timed_idea = {
						idea = GER_strengthened_supply_lines_ns
						days = 730
					}
					set_country_flag = TRM_had_strengthened_supply_lines_ns
				}
			}
			if = { # 게헨나가 "GEM_war_preparations" 중점을 완료 했고 전쟁을 선포했을때
				limit = {
					original_tag = GEM
					GEM = { 
						OR = {
							has_completed_focus = GEM_war_preparations
						}
						NOT = { has_country_flag = GEM_had_strengthened_supply_lines_ns }
					}
				}
				GEM = {
					add_timed_idea = {
						idea = GER_strengthened_supply_lines_ns
						days = 730
					}
					set_country_flag = GEM_had_strengthened_supply_lines_ns
				}
			}
		}
	}
	on_state_control_changed = {
		effect = {
			if = {
				limit = {

					}

						NOT = { is_core_of = TRM }
						BA_is_GER_core = yes
						OR = {
							is_core_of = GEM
							is_core_of = GEH
							is_core_of = HEU
						}
					}
				}
				log = "[GetDateText], [ROOT.GetName] control [FROM.FROM.GetName]"
				if = {
					limit = { 
						ROOT = { 
							check_variable = { 
								TRM_anti_geh_emotion_var = 0 
							}
						}
					}
					ROOT = {
						set_variable = { TRM_anti_geh_emotion_var = FROM.FROM }
						country_event = {
							id = TRMDIP.20
							hours = 0
						}
					}
				}
				
				FROM.FROM = {
					add_dynamic_modifier = { modifier = TRM_anti_geh_emotion }
				}
			}
		}
	}

	on_weekly_TRM = {
		effect = {
			IF = {
				limit = {
					has_completed_focus = TRN_unprovable_problem
				}
				IF = {
					limit = {
						has_country_leader = {
							character = TRN_Kirifuji_Nagisa
							ruling_only = yes
						}
						TRN_Kirifuji_Nagisa = { has_trait = TRN_nagisa_1_off }
						check_variable = { nagisa_faction_A > 49 } 
					}
					remove_country_leader_trait = TRN_nagisa_1_off
					add_country_leader_trait = TRN_nagisa_1_on
				}
			}
		}
		effect = {
			IF = {
				limit = { has_completed_focus = TRN_second_trinity }
				IF = {
					limit = {
						has_country_leader = {
							character = TRN_Misono_Mika
							ruling_only = yes
						}
						TRN_Mika_has_low_honor_trigger = yes
					}
					add_country_leader_trait = TRN_lowhonor_trait
					remove_country_leader_trait = TRN_verylowhonor_trait
				}
				IF = {
					limit = {
						has_country_leader = {
							character = TRN_Misono_Mika
							ruling_only = yes
						}
						TRN_Mika_has_very_low_honor_trigger = yes
					}
					remove_country_leader_trait = TRN_lowhonor_trait
					add_country_leader_trait = TRN_verylowhonor_trait
				}
				IF = {
					limit = {
						has_country_leader = {
							character = TRN_Misono_Mika
							ruling_only = yes
						}
						OR = {
							has_country_leader_with_trait = TRN_lowhonor_trait
							has_country_leader_with_trait = TRN_verylowhonor_trait
						}
						OR = {
							TRN_Mika_has_very_high_honor_trigger = yes
							TRN_Mika_has_high_honor_trigger = yes
							TRN_Mika_has_middle_honor_trigger = yes
						}
					}
					IF = {
						limit = { has_country_leader_with_trait = TRN_lowhonor_trait }
						remove_country_leader_trait = TRN_lowhonor_trait
					}
					else_if = {
						limit = { has_country_leader_with_trait = TRN_verylowhonor_trait }
						remove_country_leader_trait = TRN_verylowhonor_trait
					}
				}
			}
		}
	}
