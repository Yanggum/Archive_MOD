on_actions = {

	on_startup = {
		effect = {
			SHA = { scoped_play_song = flip_flap_festival }
		}

		# 오스트리아 자치특구
		effect = { 
			153 = { set_state_flag = austrian_state } # 티롤
			848 = { set_state_flag = austrian_state } # 포어아를베르크
			152 = { set_state_flag = austrian_state } # 오버외스터라이히
			976 = { set_state_flag = austrian_state } # 슈타이어마르크-케른텐

		}
	}
	#ROOT = 공격하는 쪽
	#FROM = 방어 측
	#두 국가가 서로 전쟁을 벌일 때 발동됨(on_war는 한 국가가 다른 국가와 전쟁을 벌일 때 발동되며, 먼저 평화로 가지 않는 한 다른 국가와 전쟁을 벌일 때는 다시 발동되지 않음).
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					ROOT = { original_tag = HYA }
					ROOT = { has_country_flag = HYA_focus_12_flag }
					FROM =  { original_tag = SHA }
				}
				ROOT = {
					clr_country_flag = HYA_focus_12_flag
					activate_mission = HYA_war_with_southern_kivotos_decision
					set_variable = { global.BAKSANWAR_on_war_relation_added_tag = THIS }
				}

				SHA = {
					if = {
						limit = { NOT = { has_completed_focus = SHA_invade_of_Hyakkiyako } }
						unlock_national_focus = SHA_invade_of_Hyakkiyako
					}
				}
			}
			if = {
				limit = {
					ROOT = { original_tag = SHA }
					ROOT = { has_country_flag = SHA_invade_of_Hyakkiyako_flag }
					FROM =  { original_tag = HYA }
				}
				ROOT = {
					clr_country_flag = SHA_invade_of_Hyakkiyako_flag
					activate_mission = SHA_war_with_southern_kivotos_decision
					set_variable = { global.BAKSANWAR_on_war_relation_added_tag = THIS }
				}

				HYA = {
					if = {
						limit = { NOT = { has_completed_focus = HYA_focus_12 } }
						unlock_national_focus = HYA_focus_12
					}
				}
			}
		}
	}
}
