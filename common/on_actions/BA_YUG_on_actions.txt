on_actions = {
	
	#이탈리아-유고슬라비아 전쟁 개전(이탈리아 선전포고)
	on_declare_war = {
		effect = {
			if = {
				limit = {
				tag = ITA
				    FROM = {
					tag = YUG
				    }
			}
			YUG = {
			    country_event = {
				    id = YUG_event.18
				    hours = 1
		        }
			}
		}
		}
	}

	#이탈리아-유고슬라비아 전쟁 개전(유고슬라비아 선전포고)
	on_declare_war = {
		effect = {
			if = {
				limit = {
				tag = YUG
				    FROM = {
					tag = ITA
				    }
			    }
			YUG = {
			    country_event = {
				    id = YUG_event.19
				    hours = 1
			    }
		    }
			}
		}
	}
    #이탈리아-유고슬라비아 전쟁 개전(이탈리아가 유고슬라비아의 괴뢰국인 알바니아에 선전포고)
	on_declare_war = {
		effect = {
			if = {
				limit = {
					tag = ITA
				        FROM = {
					     tag = ALB
				        }	
					ALB = {
						is_puppet_of = YUG
					}
					
			    }

			YUG = {
			    country_event = {
				    id = YUG_event.18
				    hours = 1
			    }
		    }
			}
		}
	}

	

}
