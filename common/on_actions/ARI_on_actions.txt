on_actions = {
	on_war = {
		effect = {
			if = { # 게헨나가 "GEM_war_preparations" 중점을 완료 했고 전쟁을 선포했을때
				limit = {
					original_tag = ARI
					ARI = { 
						OR = {
							has_completed_focus = ARI_sneaking_into_enemy_gaps
						}
						NOT = { has_country_flag = ARI_guerrilla_warfare_ns }
					}
				}
				ARI = {
					set_country_flag = ARI_guerrilla_warfare_ns
				}
				every_state = {
					limit = {
						BA_prussian_territory_is = yes
					}
					add_dynamic_modifier = {
						modifier = ARI_guerrilla_warfare
						days = 200
						scope = ARI
					}
				}
			}
		}
	}
	
	#ROOT = attacking side
	#FROM = defending side
	#fired when two countries end up at war with each other (on_war is fired when a country goes to war against anyone and is not fired again when it enters war against another country unless it went to peace first)
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					ROOT = { original_tag = ARI }
					NOT = { FROM = { original_tag = ARI } }
				}
				FROM = {
					add_to_array = { 
						array = ARI.ARI_war_targets_array 
						value = THIS
					}
					log = "[THIS.GetName] was added to array"
				}
			}
			if = {
				limit = {
					FROM = { original_tag = ARI }
					NOT = { ROOT = { original_tag = ARI } }
				}
				ROOT = {
					add_to_array = { 
						array = ARI.ARI_war_targets_array 
						value = THIS
					}
					log = "[THIS.GetName] was added to array"
				}
			}
		}
	}
}
