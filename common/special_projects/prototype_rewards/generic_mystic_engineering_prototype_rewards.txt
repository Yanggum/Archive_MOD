# Low Scientist XP gain (소량의 과학자 경험치)
sp_mystic_generic_reward_scientist_xp_1 = {
	fire_only_once = yes

	threshold = {
		min = 0
		max = 100
	}

	weight = {
		base = @base_reward_weight
		modifier = {
			factor = @reward_weight_factor_theme
			has_project_flag = sp_mystic_generic_reward_scientist_xp_flag
		}
		modifier = {
			factor = 0
			var:scientist = {
				has_scientist_level = {
					level > 4
					specialization = specialization_mystic_engineering
				}
			}
		}
	}
	
	option = {
		token = sp_mystic_generic_reward_option_scientist_xp
		
		iteration_output = {
			scientist_effects = {
				if = {
					limit = { }
					add_scientist_xp = {
						experience = constant:sp_scientist_xp_gain.low
						specialization = specialization_mystic_engineering
					}
				}
				FROM = { set_project_flag = sp_mystic_generic_reward_scientist_xp_flag }
			}
		}
	}
}

# Critical Failure (치명적 실패: 진행도 손실 + 시설 파괴 + 과학자 부상)
sp_mystic_generic_reward_critical_failure = {
	fire_only_once = yes
	
	threshold = {
		min = 25
		max = 80
	}

	weight = {
		base = 0.5
	}
	
	option = {
		token = sp_mystic_generic_reward_option_critical_failure
		
		iteration_output = {
			country_effects = { 
				FROM = { add_project_progress_ratio = constant:sp_progress.loss.medium }
			}

			scientist_effects = {
				add_scientist_xp = {
					experience = constant:sp_scientist_xp_gain.low
					specialization = specialization_mystic_engineering
				}
				injure_scientist_for_days = 30
			}

			facility_state_effects = {
				damage_building = {
					tags = facility
					damage = 0.25
					province = var:facility_province_id
				}
			}
		}
	}
}
