country_event = {
	id = TRM_auto_conference.1
	title = TRM_auto_conference.1.t
	desc = TRM_auto_conference.1.d
	picture = GFX_report_event_generic_sign_treaty2

	fire_only_once = yes

	trigger = {
		#A democratic faction leader whose faction holds at least one German core state while SOV holds another can get this.
		has_war_with = TRM
		NOT = { has_country_flag = GEH_unified_gehenna_flag }
		OR = {
			original_tag = GEM
			original_tag = HEU
		}
		TRM = {
			surrender_progress > 0.4
			has_government = fascism
		}
		is_faction_leader = yes
		any_state = {
			is_core_of = TRM
			controller = {
				is_in_faction_with = ROOT
			}
		}
		NOT = {
			tag = GEH
		}
		NOT = {
			has_war_with = GEH
		}
		GEH = {
			NOT = { has_country_flag = GEH_unified_gehenna_flag }
			is_in_faction_with = ROOT
			has_war_with = TRM
		}
		NOT = {
			has_global_flag = had_auto_conference_1
		}
		NOT = {
			any_state = {
				is_core_of = TRM
				controller = {
					NOT = {
						tag = TRM
					}
					is_in_faction = yes
					NOT = {
						OR = {
							is_in_faction_with = ROOT
							is_in_faction_with = COU
						}
					}
				}
			}
		}
	}

	mean_time_to_happen = {
		days = 8
	}

	immediate = {
		set_global_flag = had_auto_conference_1
		GEH = {
			save_event_target_as = auto_conference_1_partner
		}
		ROOT = {
			save_event_target_as = auto_conference_1_origin
			set_country_flag = auto_conference_1_signatory_faction_leader
		}
	}

	option = {
		name = TRM_auto_conference.1.a
		ai_chance = {
			base = 10
		}
		hidden_effect = {
			event_target:auto_conference_1_partner = {
				country_event = { days = 1 id = TRM_auto_conference.101 }
			}
		}
		custom_effect_tooltip = TRM_auto_conference_1_division_tt
		event_target:auto_conference_1_partner = {
			custom_effect_tooltip = needs_approval
		}
	}

	option = {
		name = TRM_auto_conference.1.b
		ai_chance = {
			base = 0
		}
	}
}
