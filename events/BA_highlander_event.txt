	is_triggered_only = yes

		name = Highlander.1.b
		ai_chance = {
			base = 1
			modifier = {
				tag = TRM
				add = 1
			}
		}
		add_timed_idea = {
			idea = BA_highlander_invest_small
			days = 90
		}
    }
	option = { #대규모 투자
		name = Highlander.1.f
		if = {
			limit = {
				has_country_flag = BA_highlander_discount
			}
			add_timed_idea = {
				idea = BA_highlander_invest_mid
				days = 90
			}
		}
		else = {
			add_timed_idea = {
				idea = BA_highlander_invest_big
				days = 90
			}
		}
		custom_effect_tooltip = BA_highlander_invest_big_tt
		add_to_variable = {
			BA_highlander_invest = 4
		}
		ai_chance = {
			base = 0
			modifier = {
				OR = {
					tag = MIL
					tag = GEM
					#tag = GEH
					tag = SHE
					#tag = TRM
					tag = COU
				}
				add = 1
			}
			modifier = {
				tag = TRM
				add = -1
			}
		}
    }
}
