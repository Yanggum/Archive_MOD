# 베트남이 VIN_french 태그를 가지고 있는데 FRA의 괴뢰국이 아닐 경우
country_event = {
	id = ai_events.1
	title = ai_events.1.t
	desc = ai_events.1.desc
	picture = GFX_report_event_ast_navy
	
	fire_only_once = yes
    	is_triggered_only = no

	hidden = yes

	trigger = {
		tag = VIN
		has_cosmetic_tag = VIN_french
		NOT = { is_puppet_of = FRA }
	}

	immediate = {
		drop_cosmetic_tag = yes
	}
}
