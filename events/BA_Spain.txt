country_event = { #대통령 축출
	id = SPBA_event.2
	title = SPBA_event.2.t
	desc = SPBA_event.2.desc
	picture = GFX_report_event_spr_franco_victory
	
	is_triggered_only = yes

	fire_only_once = yes

	option = {
		name = SPBA_event.2.a
		set_politics = {
			ruling_party = fascism
			elections_allowed = yes
		}
		if = {
			limit = {
				NOT = {
					has_country_leader = {
						character = SPR_indalecio_prieto
						ruling_only = yes
					}
				}
			}
			if = {
				limit = {
					NOT = {
						can_be_country_leader = SPR_indalecio_prieto
					}
				}
				add_country_leader_role = {
					character = SPR_indalecio_prieto
					country_leader = {
					 	ideology = socialism
					 	traits = {  }
					 	expire = "1965.1.1"
					}
					promote_leader = yes
				}
			}
			else = {
				promote_character = {
				    character = SPR_indalecio_prieto
				    ideology = socialism
				}
			}
		}
		set_party_name = {
			ideology = democratic
			long_name = SPR_PSOE_party_long
			name = SPR_PSOE_party
		}
		add_stability = 0.10
		add_popularity = {
			ideology = democratic
			popularity = -0.10
		}
		add_popularity = {
			ideology = communism
			popularity = -0.10
		}
	}
}
