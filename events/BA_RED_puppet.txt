	desc = redpuppet.1000.desc

	is_triggered_only = yes 

	immediate = {
		hidden_effect = {
			set_global_flag = RE4_northern_sovereignty_war_started
			RE4 = {
				RED = { end_puppet = RE4 }
				declare_war_on = {
					target = RED
					type = civil_war
				}
				add_civil_war_target = RED

				add_dynamic_modifier = {
					modifier = RE4_investment_in_finland_dynamic_modifier
				}
			}
			if = {
				limit = { original_tag = RE4 }
				RED = { country_event = redpuppet.1001 }
			}
			if = {
				limit = { original_tag = RED }
				RE4 = { country_event = redpuppet.1001 }
			}
		}
	}
	
	option = {
		name = redpuppet.1000.a
		every_other_country = {
			limit =  {
				OR = {
					original_tag = RE1
					original_tag = RE2
					original_tag = RE3
				}
				has_country_flag = RED_northern_sovereignty_war_1side_team_flag
			}
			add_war_support = 0.10
			add_to_war = {
				targeted_alliance = RED
				enemy = RE4
			}
		}
		every_other_country = {
			limit =  {
				OR = {
					original_tag = RE1
					original_tag = RE2
					original_tag = RE3
				}
				NOT = { has_country_flag = RED_northern_sovereignty_war_1side_team_flag }
			}
			RED = {
				end_puppet = PREV
			}
			RE4 = {
				set_autonomy = {
